# Q19 - backup_check.sh
cat > backup_check.sh <<'EOF'
#!/bin/bash

if [ ! -d "$1" ]; then
    echo "Directory does not exist"
    exit 1
fi

count=0

for file in "$1"/*.bak
do
    if [ -f "$file" ]; then
        echo "Backup found: $file"
        count=$((count+1))
    fi
done

echo "Total backup files: $count"

if [ "$count" -eq 0 ]; then
    echo "No backup files found" >> logs/backup_log.txt
fi
EOF
chmod +x backup_check.sh
./backup_check.sh backups


# Q20 - file_report.sh
cat > file_report.sh <<'EOF'
#!/bin/bash

if [ $# -ne 1 ]; then
    echo "Usage: $0 <filename>"
    exit 1
fi

if [ ! -f "$1" ] || [ ! -r "$1" ]; then
    echo "File does not exist or is not readable"
    exit 1
fi

wc "$1"
echo "$(date): Processed $1" >> logs/execution_history.log
EOF
chmod +x file_report.sh
./file_report.sh data/names.txt
